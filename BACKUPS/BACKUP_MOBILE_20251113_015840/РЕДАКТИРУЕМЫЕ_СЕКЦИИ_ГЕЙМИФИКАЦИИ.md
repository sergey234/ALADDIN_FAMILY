### 2. Хранение в UserDefaults

```swift
// В RewardsModalView:
@AppStorage("earning_ways_list") private var earningWaysData: String = ""
@AppStorage("punishment_reasons_list") private var punishmentReasonsData: String = ""

@State private var earningWays: [EarningWay] = []
@State private var punishmentReasons: [PunishmentReason] = []

// Функции загрузки/сохранения:
func loadEarningWays() {
    if let data = earningWaysData.data(using: .utf8),
       let decoded = try? JSONDecoder().decode([EarningWay].self, from: data) {
        earningWays = decoded
    } else {
        earningWays = EarningWay.defaultEarningWays
        saveEarningWays()
    }
}

func saveEarningWays() {
    if let encoded = try? JSONEncoder().encode(earningWays),
       let string = String(data: encoded, encoding: .utf8) {
        earningWaysData = string
        NotificationCenter.default.post(name: UserDefaults.didChangeNotification, object: nil)
    }
}

// Аналогично для punishmentReasons
```

```swift
private func earningWayEditableRow(earningWay: Binding<EarningWay>) -> some View {
    VStack(alignment: .leading, spacing: Spacing.s) {
        HStack {
            // Иконка (можно изменить)
            Text(earningWay.icon.wrappedValue)
                .font(.system(size: 24))
            
            VStack(alignment: .leading, spacing: Spacing.xs) {
                Text(earningWay.title.wrappedValue)
                    .font(.body)
                    .fontWeight(.semibold)
                    .foregroundColor(.textPrimary)
                
                Text(earningWay.subtitle.wrappedValue)
                    .font(.caption)
                    .foregroundColor(.textSecondary)
            }
            
            Spacer()
            
            // Награда (можно изменить)
            Text("+\(earningWay.amount.wrappedValue) 🦄")
                .font(.bodyBold)
                .foregroundColor(.successGreen)
        }
        
        // Панель управления (только для родителей)
        if isUserParent {
            HStack(spacing: Spacing.m) {
                // Toggle вкл/выкл
                VStack(spacing: Spacing.xxs) {
                    Toggle("", isOn: earningWay.isEnabled)
                        .labelsHidden()
                        .scaleEffect(0.8)
                    Text(earningWay.isEnabled.wrappedValue ? "Вкл" : "Выкл")
                        .font(.captionSmall)
                        .foregroundColor(earningWay.isEnabled.wrappedValue ? .successGreen : .textSecondary)
                }
                
                // Кнопка редактирования
                Button(action: {
                    editingEarningWay = earningWay.wrappedValue
                    showEditEarningWay = true
                }) {
                    HStack(spacing: Spacing.xs) {
                        Image(systemName: "pencil.circle.fill")
                            .font(.caption)
                        Text("Изменить")
                            .font(.caption)
                    }
                    .foregroundColor(.secondaryGold)
                    .padding(.horizontal, Spacing.s)
                    .padding(.vertical, Spacing.xs)
                    .background(
                        RoundedRectangle(cornerRadius: CornerRadius.small)
                            .fill(Color.secondaryGold.opacity(0.1))
                    )
                }
                
                // Быстрое изменение награды (+/-)
                HStack(spacing: Spacing.xs) {
                    Button(action: {
                        if earningWay.amount.wrappedValue > 1 {
                            earningWay.amount.wrappedValue -= 1
                            saveEarningWays()
                        }
                    }) {
                        Image(systemName: "minus.circle.fill")
                            .foregroundColor(.dangerRed)
                    }
                    
                    Text("\(earningWay.amount.wrappedValue)")
                        .font(.caption)
                        .frame(width: 30)
                    
                    Button(action: {
                        earningWay.amount.wrappedValue += 1
                        saveEarningWays()
                    }) {
                        Image(systemName: "plus.circle.fill")
                            .foregroundColor(.successGreen)
                    }
                }
                
                Spacer()
            }
        }
    }
    .padding(Spacing.m)
    .background(
        RoundedRectangle(cornerRadius: CornerRadius.medium)
            .fill(earningWay.isEnabled.wrappedValue ? Color.successGreen.opacity(0.05) : Color.backgroundMedium.opacity(0.3))
            .overlay(
                RoundedRectangle(cornerRadius: CornerRadius.medium)
                    .stroke(
                        earningWay.isEnabled.wrappedValue ? Color.successGreen.opacity(0.3) : Color.textSecondary.opacity(0.2),
                        lineWidth: 1
                    )
            )
    )
}
```

### 4. Модалы для добавления/редактирования

**Модал редактирования:**
```swift
struct EditEarningWayModal: View {
    @Binding var isPresented: Bool
    @Binding var earningWay: EarningWay
    let onSave: () -> Void
    
    @State private var icon: String = ""
    @State private var title: String = ""
    @State private var subtitle: String = ""
    @State private var amount: Int = 0
    
    var body: some View {
        NavigationView {
            Form {
                Section("Иконка") {
                    TextField("📚", text: $icon)
                }
                Section("Название") {
                    TextField("Домашнее задание", text: $title)
                }
                Section("Описание") {
                    TextField("+10 единорогов за задание", text: $subtitle)
                }
                Section("Награда (единороги)") {
                    Stepper("\(amount) 🦄", value: $amount, in: 1...1000)
                }
            }
            .navigationTitle("Редактировать")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .navigationBarTrailing) {
                    Button("Сохранить") {
                        earningWay.icon = icon
                        earningWay.title = title
                        earningWay.subtitle = subtitle
                        earningWay.amount = amount
                        onSave()
                        isPresented = false
                    }
                }
            }
        }
    }
}
```

## ✅ ФУНКЦИОНАЛЬНОСТЬ ДЛЯ РОДИТЕЛЕЙ

### В каждой редактируемой секции родители могут:

1. **Включить/выключить** любой пункт (toggle)
2. **Изменить награду/наказание** (кнопка "Изменить" → модал)
3. **Быстро изменить сумму** (+/- кнопки прямо в карточке)
4. **Добавить новый пункт** (кнопка "+ Добавить")
5. **Удалить пункт** (кнопка "Удалить" в модале редактирования)

### В секции "🏪 Управление магазином":

- Все награды видны в развернутом виде
- Toggle вкл/выкл прямо в списке
- Кнопка редактирования для каждой награды
- Кнопка "+ Добавить новую награду" (уже есть функционал)
- Кнопка "⚙️ Настроить" → открывает полный модал (как сейчас)

### В секции "🎮 Управление играми":

- Все игры видны с основными параметрами
- Показ основных наград/цен
- Toggle вкл/выкл для каждой игры (только для тех, что можно выключить)
