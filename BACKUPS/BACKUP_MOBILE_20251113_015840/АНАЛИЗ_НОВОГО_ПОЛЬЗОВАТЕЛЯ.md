## 4️⃣ РЕШЕНИЕ ПРОБЛЕМЫ

### 🔧 ПРАВИЛЬНАЯ ЛОГИКА:

#### **Вариант 1: Пустой список для нового пользователя**

```swift
@State private var familyMembers: [FamilyMemberData] = []

private func loadFamilyMembers() {
    // Загружаем из UserDefaults или API
    if let savedMembers = UserDefaults.standard.data(forKey: "family_members_list"),
       let decoded = try? JSONDecoder().decode([FamilyMemberData].self, from: savedMembers) {
        familyMembers = decoded
    } else {
        // Для нового пользователя - показываем только его самого
        if let currentUserRole = UserDefaults.standard.string(forKey: "current_user_role"),
           let role = FamilyMemberCard.FamilyRole(rawValue: currentUserRole) {
            let currentUserName = UserDefaults.standard.string(forKey: "current_user_name") ?? "Вы"
            familyMembers = [
                FamilyMemberData(
                    name: currentUserName,
                    role: role,
                    avatar: getAvatarForRole(role),
                    status: .protected,
                    threatsBlocked: 0,
                    lastActive: "Сейчас"
                )
            ]
        } else {
            // Если вообще нет данных - пустой список
            familyMembers = []
        }
    }
    print("✅ Загружено участников семьи: \(familyMembers.count)")
}
```

#### **Вариант 2: Показывать только реальных участников из API**

```swift
private func loadFamilyMembers() {
    // Загрузка из API (FamilyViewModel)
    familyViewModel.loadFamilyMembers()
    
    // Отображаем только реальных участников
    familyMembers = familyViewModel.familyMembers.map { member in
        FamilyMemberData(
            name: member.name,
            role: FamilyMemberCard.FamilyRole(rawValue: member.role) ?? .parent,
            avatar: member.avatar,
            status: FamilyMemberCard.ProtectionStatus(rawValue: member.status) ?? .protected,
            threatsBlocked: member.threatsBlocked,
            lastActive: member.lastActive
        )
    }
}
```
