private func loadChildren() {
    guard let savedData = UserDefaults.standard.data(forKey: "family_members_list"),
          let decoded = try? JSONDecoder().decode([FamilyMemberData].self, from: savedData) else {
        children = []
        return
    }

        .filter { member in
            member.role == .child || member.role == .teenager
        }
        .map { $0.name }
}

## ⚠️ ЧТО НУЖНО ДОРАБОТАТЬ

### 1. Загрузка детей из API

**Проблема:**
- Дети загружаются из `UserDefaults`, но в `.onAppear` модала нет вызова `loadChildren()`
- Нет прямого вызова API для загрузки детей в этом модале

**Решение:**
Добавить в `.onAppear` модала:

```swift
.onAppear {
    // Загружаем детей из API
    loadChildrenFromAPI()
    
    // Применяем правила при открытии модала (если уже включены)
    if isAutomatedRulesEnabled {
        applyAgeBasedRules()
    }
}

private func loadChildrenFromAPI() {
    let apiService = APIService.shared
    apiService.getFamilyMembers { result in
        DispatchQueue.main.async {
            switch result {
            case .success(let members):
                // Фильтруем только детей
                let childMembers = members.filter { member in
                    let role = member.role.lowercased()
                    return role == "child" || role == "teenager"
                }
                // Сохраняем в UserDefaults для быстрого доступа
                if let encoded = try? JSONEncoder().encode(childMembers) {
                    UserDefaults.standard.set(encoded, forKey: "family_members_list")
                }
                // Обновляем список детей
                self.children = childMembers.map { $0.name }
                
                // Если выбранный ребёнок не найден, выбираем первого
                if !self.children.contains(self.selectedChild) && !self.children.isEmpty {
                    self.selectedChild = self.children.first ?? ""
                }
            case .failure(let error):
                print("⚠️ Failed to load children: \(error.localizedDescription)")
                // Fallback: загружаем из UserDefaults
                loadChildren()
            }
        }
    }
}
```
