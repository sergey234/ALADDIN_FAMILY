**Создать файл:** `Core/Performance/PerformanceProfiler.swift`

```swift
import Foundation
import os.signpost

/// Профилирование производительности
class PerformanceProfiler {
    static let shared = PerformanceProfiler()
    
    private let log = OSLog(subsystem: "com.aladdin.performance", category: "performance")
    
    func startMeasure(name: String) -> String {
        let signpostID = OSSignpostID(log: log)
        os_signpost(.begin, log: log, name: name, signpostID: signpostID)
        return name
    }
    
    func endMeasure(name: String) {
        os_signpost(.end, log: log, name: name)
    }
    
    func measure<T>(_ name: String, block: () throws -> T) rethrows -> T {
        let signpostID = OSSignpostID(log: log)
        os_signpost(.begin, log: log, name: name, signpostID: signpostID)
        defer {
            os_signpost(.end, log: log, name: name, signpostID: signpostID)
        }
        return try block()
    }
    
    func measureAsync<T>(_ name: String, block: () async throws -> T) async rethrows -> T {
        let signpostID = OSSignpostID(log: log)
        os_signpost(.begin, log: log, name: name, signpostID: signpostID)
        defer {
            os_signpost(.end, log: log, name: name, signpostID: signpostID)
        }
        return try await block()
    }
}
```

**Использование:**
```swift
// В ViewModels
func load() async {
    await PerformanceProfiler.shared.measureAsync("LoadAnalytics") {
        // Загрузка данных
    }
}
```

func request(_ endpoint: String) async throws -> Data {
    let startTime = Date()
    
    // Логирование запроса
    print("📡 Запрос: \(endpoint)")
    
    let data = try await performRequest(endpoint)
