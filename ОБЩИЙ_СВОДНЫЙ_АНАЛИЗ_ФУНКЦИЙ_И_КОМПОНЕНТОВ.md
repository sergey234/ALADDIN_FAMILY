extension ComponentStatusService {
    func enableComponentsForTariff(_ tariffType: TariffType) {
        // Получить список компонентов доступных для тарифа
        let availableComponents = getAvailableComponents(for: tariffType)
        
        // Активировать каждый компонент
        availableComponents.forEach { componentId in
            enableComponent(componentId)
        }
    }
    
    private func getAvailableComponents(for tariffType: TariffType) -> [String] {
        // Логика определения доступных компонентов по тарифу
        switch tariffType {
        case .free: return ["basic_components"]
        case .personal: return ["basic_components", "personal_components"]
        case .family: return ["basic_components", "personal_components", "family_components"]
        case .premium: return ["all_components"]
        }
    }
}
