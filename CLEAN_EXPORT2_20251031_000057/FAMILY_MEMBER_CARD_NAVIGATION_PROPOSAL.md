## 🎯 Предлагаемое решение

### 1. Создать NavigationManager для карточек

```swift
// Core/Navigation/FamilyNavigationManager.swift

class FamilyNavigationManager: ObservableObject {
    
    func navigateToMember(member: FamilyMember) {
        switch member.role {
        case .parent:
            // Переход на ParentalControlScreen
            navigationManager.navigate(to: .parentalControl)
            
        case .child:
            // Переход на ChildInterfaceScreen
            navigationManager.navigate(to: .childInterface)
            
        case .teenager:
            // Упрощенная версия ParentalControl
            navigationManager.navigate(to: .teenagerInterface)
            
        case .elderly:
            // Переход на ElderlyInterfaceScreen
            navigationManager.navigate(to: .elderlyInterface)
        }
    }
}
```

### 2. Обновить FamilyMemberCard

```swift
// Добавить параметр navigation
```
