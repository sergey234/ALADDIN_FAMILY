### **ШАГ 4: Получение токена с TRIAL**

**Код:**

// Core/Managers/SubscriptionManager.swift (строки 652-755)
func registerDeviceAnonymously() async throws -> JWTToken {
    let deviceId = UIDevice.current.identifierForVendor?.uuidString ?? UUID().uuidString
    let deviceType = "ios"
    
    let request = DeviceRegisterRequest(deviceId: deviceId, deviceType: deviceType)
    
    // ✅ Вызов API БЕЗ авторизации
    let response = try await APIService.shared.registerDeviceAnonymously(request: request)
    
    // ✅ Создаем токен из ответа
    let jwtToken = JWTToken(
        token: response.token,
        deviceId: response.deviceId,
        subscriptionLevel: .trial,  // ✅ TRIAL автоматически!
        trialInfo: response.subscription.trialInfo,
        expiresAt: response.expiresAtDate ?? Date().addingTimeInterval(86400)
    )
    
    // ✅ Сохраняем токен в Keychain
    await storeToken(jwtToken)
    
    return jwtToken
}
```

**Что происходит:**
- Приложение отправляет запрос: `POST /api/auth/register-device`
- Сервер создает токен **БЕЗ персональных данных**
- Сервер возвращает токен с TRIAL подпиской (14 дней)
- Токен сохраняется в Keychain

// Core/Managers/SubscriptionManager.swift (строки 492-548)
func activateTrialIfNeeded() async {
    // Проверяем, не использован ли уже TRIAL
    let hasUsedTrial = UserDefaults.standard.bool(forKey: "trial_used")
    if hasUsedTrial {
        return  // TRIAL уже использован
    }
    
    // Проверяем, активен ли уже TRIAL
    if let trial = trialStatus, trial.isActive {
        return  // TRIAL уже активен
    }
    
    logger.business("🎁 Activating 14-day trial period")
    
    // ✅ Вычисляем даты TRIAL
    let startDate = Date()
    let endDate = Calendar.current.date(byAdding: .day, value: 14, to: startDate)!
    
    let trialInfo = TrialInfo(
        startDate: startDate,
        endDate: endDate,
        durationDays: 14
    )
    
    // ✅ Регистрируем устройство
    try await registerDeviceAnonymously()
    
    // ✅ Активируем TRIAL
    await updateTrialStatus(trialInfo)
    
    // ✅ Помечаем TRIAL как использованный
    UserDefaults.standard.set(true, forKey: "trial_used")
    
    logger.business("✅ Trial activated successfully: 14 days remaining")
}
```

**Что происходит:**
- Приложение автоматически активирует TRIAL (14 дней)
- Все функции работают как в платном тарифе
- Пользователь получает полный доступ на 14 дней
