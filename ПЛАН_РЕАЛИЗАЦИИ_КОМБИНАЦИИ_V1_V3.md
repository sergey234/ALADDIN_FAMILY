## ЭТАП 2: Функции загрузки и сохранения 💾

### Задача 2.1: Создать функцию loadFamilyMembers()
**Файл:** `Screens/02_FamilyScreen.swift`
**Где:** После функции `navigateToMemberScreen()`

**Логика:**
1. Попытаться загрузить из UserDefaults
2. Если нет данных - создать карточку текущего пользователя
3. Если нет даже роли - оставить пустым

**Код:**
```swift
private func loadFamilyMembers() {
    // 1. Попытка загрузить из UserDefaults
    if let savedData = UserDefaults.standard.data(forKey: familyMembersKey),
       let decoded = try? JSONDecoder().decode([FamilyMemberData].self, from: savedData),
       !decoded.isEmpty {
        familyMembers = decoded
        print("✅ Загружено участников семьи из UserDefaults: \(familyMembers.count)")
        return
    }
    
    // 2. Если нет сохранённых данных - создать карточку текущего пользователя
    if let currentRoleString = UserDefaults.standard.string(forKey: currentUserRoleKey),
       let role = FamilyMemberCard.FamilyRole(rawValue: currentRoleString) {
        let userName = UserDefaults.standard.string(forKey: currentUserNameKey) ?? "Вы"
        
        familyMembers = [
            FamilyMemberData(
                name: userName,
                role: role,
                avatar: getAvatarForRole(role),
                status: .protected,
                threatsBlocked: 0,
                lastActive: "Сейчас"
            )
        ]
        print("✅ Создан участник для текущего пользователя: \(userName)")
    } else {
        // 3. Если даже роли нет - пустой список
        familyMembers = []
        print("⚠️ Нет данных о пользователе, список пуст")
    }
    
    // Сохраняем созданный список
    saveFamilyMembers()
}
```

### Задача 2.2: Создать функцию saveFamilyMembers()
**Файл:** `Screens/02_FamilyScreen.swift`
**Где:** После `loadFamilyMembers()`

**Код:**
```swift
private func saveFamilyMembers() {
    guard let encoded = try? JSONEncoder().encode(familyMembers) else {
        print("❌ Ошибка кодирования участников семьи")
        return
    }
    
    UserDefaults.standard.set(encoded, forKey: familyMembersKey)
    print("✅ Сохранено участников семьи: \(familyMembers.count)")
    
    // Уведомляем другие экраны об изменении
    NotificationCenter.default.post(name: UserDefaults.didChangeNotification, object: nil)
}
```

### Задача 3.1: Создать computed property isFamilyCreator

```swift
// Определяем, является ли пользователь создателем семьи
private var isFamilyCreator: Bool {
    let familyId = UserDefaults.standard.string(forKey: familyIdKey)
    let memberCount = familyMembers.count
    
    // Если нет familyID или участников <= 1 - вероятно создатель
    // Также проверяем, если в списке только текущий пользователь
    if familyId == nil {
        return true
    }
    
    if memberCount <= 1 {
        // Проверяем, есть ли в списке только текущий пользователь
        if let firstMember = familyMembers.first,
           firstMember.name == (UserDefaults.standard.string(forKey: currentUserNameKey) ?? "Вы") {
            return true
        }
    }
    
    return false
}
```

```swift
if let role = selectedRole,
   let userName = UserDefaults.standard.string(forKey: "current_user_name"),
   let savedData = UserDefaults.standard.data(forKey: "family_members_list"),
   var members = try? JSONDecoder().decode([FamilyMemberData].self, from: savedData) {
    
    let newMember = FamilyMemberData(
        name: userName,
        role: FamilyMemberCard.FamilyRole(rawValue: role.rawValue) ?? .parent,
        avatar: getAvatarForRole(role),
        status: .protected,
        threatsBlocked: 0,
        lastActive: "Сейчас"

            UserDefaults.standard.set(encoded, forKey: "family_members_list")
            NotificationCenter.default.post(name: UserDefaults.didChangeNotification, object: nil)
        }
    }
}
```
