```swift
func fetchSummary(period: String, filters: AnalyticsFilters) async throws -> (AnalyticsSummary, DataSource) {
    let cacheKey = "summary_\(period)_\(filters.onlyBlocked)_\(filters.includeFamily)_\(filters.includeDevices)"
    
    return try await withCheckedThrowingContinuation { continuation in
        apiService.getAnalytics(period: period) { result in
            switch result {
            case .success(let analyticsResponse):
                // ✅ Реальные данные из API
                let summary = AnalyticsSummary(...)
                self.setCachedSummary(summary, for: cacheKey)
                continuation.resume(returning: (summary, .api))
                
            case .failure(let error):
                // ✅ Проверяем кэш
                if let cachedSummary = self.getCachedSummary(for: cacheKey) {
                    continuation.resume(returning: (cachedSummary, .cache))
                    return
                }
                
                // ✅ Нет данных - показываем 0
                let emptySummary = AnalyticsSummary(
                    threatsDetected: 0,
                    threatsBlocked: 0,
                    itemsScanned: 0,
                    protectionLevel: 0.0
                )
                continuation.resume(returning: (emptySummary, .empty))
            }
        }
    }
}
```

```swift
func load() async {
    isLoading = true
    errorMessage = nil
    
    do {
        async let summaryTask = service.fetchSummary(period: defaultPeriod, filters: defaultFilters)
        async let securityTask = service.fetchSecurityAnalytics(period: defaultPeriod)
        
        let (summaryResult, securityResult) = try await (summaryTask, securityTask)
        
        // Извлекаем данные и источник
        let (summary, summarySource) = summaryResult
        let (security, securitySource) = securityResult
        
        // Определяем общий источник данных
        dataSource = summarySource == .api && securitySource == .api ? .api :
                     summarySource == .cache || securitySource == .cache ? .cache : .empty
        
        apply(summary: summary)
        apply(securityAnalytics: security)
        
    } catch {
        // Обработка ошибок
        errorMessage = getErrorMessage(from: error)
        dataSource = .error
    }
    
    isLoading = false
}
```

```swift
func getComponentStats(componentId: String, completion: @escaping (Result<ComponentStats, Error>) -> Void) {
    let endpoint: String
    switch componentId {
    case "driving": endpoint = AppConfig.Endpoint.drivingStats
    case "darkweb": endpoint = AppConfig.Endpoint.darkWebStats
    case "identity": endpoint = AppConfig.Endpoint.identityTheftStats
    case "location": endpoint = AppConfig.Endpoint.locationStats
    case "cleanup": endpoint = AppConfig.Endpoint.dataCleanupStats
    case "tracker": endpoint = AppConfig.Endpoint.antiTrackerStats
    case "ai": endpoint = AppConfig.Endpoint.aiCategoriesStats
    default: completion(.failure(NSError(domain: "APIService", code: -1, userInfo: [NSLocalizedDescriptionKey: "Unknown component"]))); return
    }
    
    networkManager.get(endpoint: endpoint, completion: completion)
}
```
