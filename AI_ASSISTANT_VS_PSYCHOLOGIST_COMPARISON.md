private func sendMessage() {
    let context = determineMessageContext(messageText)
    
    // Если психологический контекст → Psychological Support Agent
    if context == "psychological_support" || context == "crisis" {
        sendPsychologicalSupportMessage(messageText)
    }
    // Если кризис → немедленная кризисная поддержка
    else if context == "crisis" {
        sendCrisisSupport(messageText)
    }
    // Иначе → обычный AI Assistant
    else {
        sendRegularMessage(messageText, context: context)
    }
}
