### **ИСПРАВЛЕНИЕ 3: Ошибка с EmailStr**

**Файл:** `app/security/api/routers/dark_web_monitoring_router.py`

**Было:**
```python
from pydantic import EmailStr  # ❌ Требует пакет email-validator

class CheckEmailRequest(BaseModel):
    email: EmailStr
```

**Стало:**
```python
from pydantic import BaseModel, validator

class CheckEmailRequest(BaseModel):
    email: str  # ✅ Обычный str
    
    @validator('email')
    def validate_email(cls, v):
        if '@' not in v:
            raise ValueError('Invalid email format')
        return v
```

**Результат:** ✅ Не требует дополнительных пакетов

## 📊 ТЕКУЩАЯ ЛОГИКА РАБОТЫ

### **1. iOS: Загрузка данных**

**Файл:** `ViewModels/DarkWebMonitoringViewModel.swift`

**Логика:**
```swift
func loadData(status: String? = nil, severity: String? = nil) async {
    isLoading = true
    errorMessage = nil
    
    do {
        // Загружаем статистику, утечки и сканирования параллельно
        async let statsTask = apiService.getDarkWebStats()
        async let leaksTask = apiService.getDarkWebLeaks(status: status, severity: severity)
        async let scansTask = apiService.getDarkWebScans(limit: 20)
        
        let (stats, leaks, scans) = try await (statsTask, leaksTask, scansTask)
        self.stats = stats
        self.leaks = leaks
        self.scans = scans
        errorMessage = nil
        
    } catch {
        let networkError = NetworkError.from(error)
        
        // Не показываем ошибку для 404 (нет данных - это нормально)
        if case .notFound = networkError {
            self.stats = nil
            self.leaks = []
            self.scans = []
            errorMessage = nil
            return
        }
        
        // Показываем ошибку только для реальных проблем
        if networkError.isCritical {
            errorMessage = String(format: "Ошибка загрузки: %@", networkError.localizedDescription)
        }
        
        self.stats = nil
        self.leaks = []
        self.scans = []
    }
    
    isLoading = false
}
```

```swift
catch {
    let networkError = NetworkError.from(error)
    
    // Не показываем ошибку для 404 (нет данных - это нормально)
    if case .notFound = networkError {
        self.stats = nil
        self.data = []
        errorMessage = nil
        return
    }
    
    // Показываем ошибку только для реальных проблем
    if networkError.isCritical {
        errorMessage = String(format: "Ошибка загрузки: %@", networkError.localizedDescription)
    }
}
```

func getComponentStats(componentId: String, completion: @escaping (Result<ComponentStats, Error>) -> Void) {
    let endpoint: String
    switch componentId {
    case "driving_reports_agent", "driving":
        endpoint = AppConfig.Endpoint.drivingStats
    case "dark_web_monitoring_agent", "darkweb":
        endpoint = AppConfig.Endpoint.darkWebStats
    case "russian_identity_theft_protection_agent", "identity":
        endpoint = AppConfig.Endpoint.identityTheftStats
    case "location_bubble_agent", "location":
        endpoint = AppConfig.Endpoint.locationStats
    case "personal_data_cleanup_agent", "cleanup":
        endpoint = AppConfig.Endpoint.dataCleanupStats
    case "anti_tracker_agent", "tracker":
        endpoint = AppConfig.Endpoint.antiTrackerStats
    case "ai_categories_agent", "ai":
        endpoint = AppConfig.Endpoint.aiCategoriesStats
    default:
        completion(.failure(NSError(...)))
        return
    }
    
    networkManager.get(endpoint: endpoint) { (result: Result<[String: AnyCodable], Error>) in
        switch result {
        case .success(let jsonDict):
            var plainDict: [String: Any] = [:]
            for (key, value) in jsonDict {
                plainDict[key] = value.value
            }
            let stats = self.parseComponentStats(componentId: componentId, json: plainDict)
            completion(.success(stats))
        case .failure(let error):
            completion(.failure(error))
        }
    }
}
```

func parseComponentStats(componentId: String, json: [String: Any]) -> ComponentStats {
    var metrics: [String: String] = [:]
    
    switch componentId {
    case "dark_web_monitoring_agent", "darkweb":
        metrics["leaks_found"] = "\(json["totalLeaks"] as? Int ?? 0)"
        metrics["new_leaks"] = "\(json["newLeaks"] as? Int ?? 0)"
        metrics["resolved_leaks"] = "\(json["resolvedLeaks"] as? Int ?? 0)"
        metrics["critical_leaks"] = "\(json["criticalLeaks"] as? Int ?? 0)"
        
    case "russian_identity_theft_protection_agent", "identity":
        metrics["attempts"] = "\(json["totalAttempts"] as? Int ?? 0)"
        metrics["blocked"] = "\(json["blockedAttempts"] as? Int ?? 0)"
        metrics["suspicious"] = "\(json["suspiciousActivities"] as? Int ?? 0)"
        
    case "location_bubble_agent", "location":
        metrics["blocked"] = "\(json["blockedRequests"] as? Int ?? 0)"
        metrics["allowed"] = "\(json["allowedRequests"] as? Int ?? 0)"
        metrics["modified"] = "\(json["modifiedRequests"] as? Int ?? 0)"
        metrics["accuracy"] = (json["currentAccuracy"] as? String) ?? "medium"
        
    case "personal_data_cleanup_agent", "cleanup":
        let totalFreed = json["totalFreed"] as? Int ?? 0
        let freedGB = Double(totalFreed) / 1_000_000_000.0
        metrics["freed_space_gb"] = String(format: "%.1f", freedGB)
        if let lastCleanupDate = json["lastCleanupDate"] as? String {
            // Вычисляем hours ago
            metrics["last_cleanup_hours_ago"] = "0"
        }
        
    case "anti_tracker_agent", "tracker":
        metrics["blocked_total"] = "\(json["totalBlocked"] as? Int ?? 0)"
        metrics["blocked_this_week"] = "\(json["blockedThisWeek"] as? Int ?? 0)"
        metrics["effectiveness"] = "\(json["effectiveness"] as? Double ?? 0.0)"
        
    case "ai_categories_agent", "ai":
        metrics["categorized"] = "\(json["totalCategorized"] as? Int ?? 0)"
        metrics["blocked"] = "\(json["totalBlocked"] as? Int ?? 0)"
        metrics["accuracy"] = "\(json["accuracy"] as? Double ?? 0.0)"
        
    default:
        break
    }
    
    return ComponentStats(
        componentId: componentId,
        metrics: metrics,
        dataSource: .api
    )
}
```
