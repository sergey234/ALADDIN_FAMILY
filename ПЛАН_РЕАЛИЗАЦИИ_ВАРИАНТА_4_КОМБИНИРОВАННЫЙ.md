## 📊 АРХИТЕКТУРА РЕШЕНИЯ

### **Логика работы:**

```
1. Пытаемся получить данные из API
   ├─ Успех → Показываем реальные данные + кэшируем
   ├─ Ошибка → Проверяем кэш
   │   ├─ Есть кэш → Показываем кэшированные данные + индикатор "Кэш"
   │   └─ Нет кэша → Показываем 0 + индикатор "Нет данных"
   └─ Ошибка авторизации → Показываем ошибку 401
```

### **Индикаторы источника данных:**

- 🟢 **"Реальные данные"** - данные из API (актуальные)
- 🟡 **"Данные из кэша"** - кэшированные данные (могут быть устаревшими)
- ⚪ **"Нет данных"** - показываем 0 (новый пользователь или нет истории)
- 🔴 **"Ошибка"** - ошибка API (401, 500 и т.д.)

```python
from fastapi import APIRouter, Depends, Query
from app.security.jwt_auth import get_current_user

router = APIRouter(prefix="/api", tags=["analytics"])

@router.get("/analytics")
async def get_analytics(
    period: str = Query("day", regex="^(day|week|month)$"),
    current_user: dict = Depends(get_current_user)
):
    user_id = current_user.get("sub")
    
    # Получаем реальные данные из БД
    threats_detected = await db.get_threats_detected(user_id, period) or 0
    threats_blocked = await db.get_threats_blocked(user_id, period) or 0
    items_scanned = await db.get_items_scanned(user_id, period) or 0
    
    # Вычисляем уровень защиты
    protection_level = 0
    if threats_detected > 0:
        protection_level = int((threats_blocked / threats_detected) * 100)
    
    # Получаем данные по типам угроз
    threats_by_type = await db.get_threats_by_type(user_id, period) or []
    
    return {
        "period": period,
        "threatsDetected": threats_detected,
        "threatsBlocked": threats_blocked,
        "itemsScanned": items_scanned,
        "protectionLevel": protection_level,
        "topThreats": top_threats or [],
        "threatsByType": threats_by_type or []
    }
```

```swift
func fetchSummary(period: String, filters: AnalyticsFilters) async throws -> (AnalyticsSummary, DataSource) {
    let cacheKey = "summary_\(period)_\(filters.onlyBlocked)_\(filters.includeFamily)_\(filters.includeDevices)"
    
    return try await withCheckedThrowingContinuation { continuation in
        apiService.getAnalytics(period: period) { result in
            switch result {
            case .success(let analyticsResponse):
                // ✅ Реальные данные из API
                let summary = AnalyticsSummary(...)
                self.setCachedSummary(summary, for: cacheKey)
                continuation.resume(returning: (summary, .api))
                
            case .failure(let error):
                // ✅ Проверяем кэш
                if let cachedSummary = self.getCachedSummary(for: cacheKey) {
                    continuation.resume(returning: (cachedSummary, .cache))
                    return
                }
                
                // ✅ Нет данных - показываем 0
                let emptySummary = AnalyticsSummary(
                    threatsDetected: 0,
                    threatsBlocked: 0,
                    itemsScanned: 0,
                    protectionLevel: 0.0
                )
                continuation.resume(returning: (emptySummary, .empty))
            }
        }
    }
}
```

```swift
        // Определяем общий источник данных
```

```swift
func fetchComponentStats(componentId: String) async throws -> (ComponentStats, DataSource) {
    let cacheKey = "component_\(componentId)"
    
    return try await withCheckedThrowingContinuation { continuation in
        apiService.getComponentStats(componentId: componentId) { result in
            switch result {
            case .success(let stats):
                // ✅ Реальные данные из API
                self.setCachedComponentStats(stats, for: cacheKey)
                continuation.resume(returning: (stats, .api))
                
            case .failure(let error):

                    return
                }
                
                // ✅ Нет данных - показываем 0
                let emptyStats = ComponentStats(
                    componentId: componentId,
                    metrics: getEmptyMetrics(for: componentId),
                    dataSource: .empty
                )
                continuation.resume(returning: (emptyStats, .empty))
            }
        }
    }
}

func fetchAllComponentsStats() async throws -> (ComponentsAnalytics, DataSource) {
    // Параллельная загрузка всех компонентов
    async let driving = fetchComponentStats(componentId: "driving_reports_agent")
    async let darkWeb = fetchComponentStats(componentId: "dark_web_monitoring_agent")
    async let identityTheft = fetchComponentStats(componentId: "russian_identity_theft_protection_agent")
    async let location = fetchComponentStats(componentId: "location_bubble_agent")
    async let cleanup = fetchComponentStats(componentId: "personal_data_cleanup_agent")
    async let tracker = fetchComponentStats(componentId: "anti_tracker_agent")
    async let ai = fetchComponentStats(componentId: "ai_categories_agent")
    
    let results = try await (driving, darkWeb, identityTheft, location, cleanup, tracker, ai)
    
    // Определяем общий источник данных
    let sources = [results.0.1, results.1.1, results.2.1, results.3.1, results.4.1, results.5.1, results.6.1]
    let overallSource = sources.contains(.api) ? .api :
                        sources.contains(.cache) ? .cache : .empty
    
    let analytics = ComponentsAnalytics(
        drivingReports: results.0.0,
        darkWeb: results.1.0,
        identityTheft: results.2.0,
        locationBubble: results.3.0,
        dataCleanup: results.4.0,
        antiTracker: results.5.0,
        aiCategories: results.6.0
    )
    
    return (analytics, overallSource)
}
```

```swift
private func componentReportCard(
    componentId: String,
    icon: String,
    titleKey: String,
    metrics: [(String, String)],

    badgeCount: Int?,
    onTap: @escaping () -> Void
) -> some View {

    let realMetrics = getRealMetrics(for: componentId)
    let realBadgeCount = getRealBadgeCount(for: componentId)
    
    ComponentReportCard(

        icon: icon,
        title: localizationManager.localized(titleKey),
        metrics: realMetrics ?? metrics, // Используем реальные данные или fallback
        color: color,
        badgeCount: realBadgeCount ?? badgeCount,
        onTap: onTap
    )

private func getRealMetrics(for componentId: String) -> [(String, String)]? {
```
