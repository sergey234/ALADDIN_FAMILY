```swift
private func navigateToMemberScreen(role: FamilyMemberCard.FamilyRole) {
    print("🔍 DEBUG: navigateToMemberScreen вызван с role: \(role)")
    
    // Haptic feedback
    let generator = UIImpactFeedbackGenerator(style: .medium)
    generator.impactOccurred()
    
    // Navigate based on role
    switch role {
    case .parent:
        print("🔍 DEBUG: Переход к .parentalControl")
        navigationManager.navigateTo(.parentalControl)  // ⚠️ БЫЛО .parentalControl
        print("🔍 DEBUG: Текущий экран: \(navigationManager.currentScreen)")
    case .child:
        navigationManager.navigateTo(.childInterface)
    case .teenager:
        navigationManager.navigateTo(.childInterface)
    case .elderly:
        navigationManager.navigateTo(.elderlyInterface)
    }
}
```

### ✅ ЧТО ОДИНАКОВО ВО ВСЕХ ВЕРСИЯХ:

1. **НЕТ `DispatchQueue.main.async`**
   - Все версии используют прямой вызов `navigationManager.navigateTo(...)`
   - NavigationManager помечен `@MainActor`, поэтому обертка не нужна

2. **Одинаковая структура функции:**
   - Haptic feedback
   - Switch по роли
   - Прямой вызов навигации
   - Debug логирование

   ```swift

       ...
   ```
