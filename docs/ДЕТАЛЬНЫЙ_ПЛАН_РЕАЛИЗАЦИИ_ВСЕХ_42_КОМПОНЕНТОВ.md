### 📱 НОВЫЙ ЭКРАН: "Экстренная помощь"

**Файл:** `Screens/EmergencyAssistanceScreen.swift`

**Что делает:**
- Центральный экран для всех экстренных функций
- Объединяет crash_detection, roadside_assistance, emergency_response

**Как выглядит в UI:**

```swift
struct EmergencyAssistanceScreen: View {
    @StateObject private var viewModel = EmergencyAssistanceViewModel()
    
    var body: some View {
        ZStack {
            LinearGradient.backgroundGradient
                .ignoresSafeArea()
            
            ScrollView {
                VStack(spacing: 20) {
                    // Заголовок
                    ALADDINNavigationBar(
                        title: "Экстренная помощь",
                        subtitle: "Быстрая помощь в критических ситуациях"
                    )
                    
                    // Карточка: Обнаружение аварий
                    EmergencyCard(
                        icon: "🚗",
                        title: "Обнаружение аварий",
                        description: "Автоматическое обнаружение аварий и вызов помощи",
                        isEnabled: $viewModel.crashDetectionEnabled,
                        onToggle: { viewModel.toggleCrashDetection() }
                    )
                    
                    // Карточка: Помощь на дороге
                    EmergencyCard(
                        icon: "🛠️",
                        title: "Помощь на дороге",
                        description: "Вызов помощи при поломке автомобиля",
                        isEnabled: $viewModel.roadsideAssistanceEnabled,
                        onToggle: { viewModel.toggleRoadsideAssistance() }
                    )
                    
                    // Карточка: Экстренные контакты
                    EmergencyContactsCard(
                        contacts: $viewModel.emergencyContacts,
                        onAdd: { viewModel.showAddContact = true }
                    )
                    
                    // История запросов
                    EmergencyHistoryCard(
                        history: $viewModel.emergencyHistory
                    )
                }
                .padding()
            }
        }
    }
}
```
