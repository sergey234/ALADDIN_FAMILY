## 🔍 КАК ЭТО РАБОТАЕТ

### 1. AsyncSemaphore (асинхронный семафор)

```swift
actor AsyncSemaphore {
    private var count: Int  // Количество доступных слотов
    private var waiters: [CheckedContinuation<Void, Never>] = []  // Очередь ожидающих
    
    func wait() async {
        if count > 0 {
            count -= 1  // Занимаем слот
        } else {
            // Ждем освобождения слота
            await withCheckedContinuation { continuation in
                waiters.append(continuation)
            }
        }
    }
    
    func signal() {
        if let waiter = waiters.first {
            waiters.removeFirst()
            waiter.resume()  // Освобождаем слот для ожидающего
        } else {
            count += 1  // Освобождаем слот
        }
    }
}
```

### 2. ParallelLoader.executeWithLimit()

```swift
// 1. Сортируем по приоритету
let sortedItems = items.sorted { $0.priority < $1.priority }

// 2. Создаем семафор с лимитом
let semaphore = AsyncSemaphore(count: maxConcurrent)

// 3. Для каждого элемента:
for item in sortedItems {
    await semaphore.wait()  // Ждем доступный слот
    
    group.addTask {
        defer { await semaphore.signal() }  // Освобождаем слот после завершения
        return try await item.loadTask()
    }
}
```
