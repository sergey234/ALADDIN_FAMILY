struct ChildProtectionSection: View {
    @StateObject private var viewModel = ChildProtectionViewModel()
    
    var body: some View {
        VStack(alignment: .leading, spacing: 16) {
            Text("Защита детей")
                .font(.system(size: 24, weight: .bold))
                .foregroundColor(.white)
                .padding(.horizontal)
            
            // Обнаружение самоповреждения
            ChildProtectionCard(
                icon: "⚠️",
                title: "Обнаружение самоповреждения",
                description: "Обнаруживает признаки самоповреждения",
                isEnabled: $viewModel.selfHarmDetectionEnabled,
                onToggle: { viewModel.toggleSelfHarmDetection() }
            )
            
            // Обнаружение груминга
            ChildProtectionCard(
                icon: "🛡️",
                title: "Обнаружение груминга",
                description: "Обнаруживает попытки обольщения",
                isEnabled: $viewModel.groomingDetectionEnabled,
                onToggle: { viewModel.toggleGroomingDetection() }
            )
            
            // Защита от онлайн-хищников
            ChildProtectionCard(
                icon: "👮",
                title: "Защита от онлайн-хищников",
                description: "Блокирует подозрительных пользователей",
                isEnabled: $viewModel.onlinePredatorsEnabled,
                onToggle: { viewModel.toggleOnlinePredators() }
            )
            
            // Психологическая поддержка
            ChildProtectionCard(
                icon: "💚",
                title: "Психологическая поддержка",

        }
    }
