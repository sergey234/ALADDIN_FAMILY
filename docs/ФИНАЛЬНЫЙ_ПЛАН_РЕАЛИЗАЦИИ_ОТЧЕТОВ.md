```swift
struct ComponentReportCard: View {
    let componentId: String
    let icon: String
    let title: String
    let metrics: [(String, String)]
    let color: Color
    let badgeCount: Int? // Количество новых событий
    let onTap: () -> Void
    
    var body: some View {
        Button(action: onTap) {
            HStack(spacing: Spacing.m) {
                // Иконка с badge
                ZStack {
                    Circle()
                        .fill(color.opacity(0.2))
                        .frame(width: 50, height: 50)
                    
                    Text(icon)
                        .font(.system(size: 24))
                    
                    // Badge для новых событий
                    if let count = badgeCount, count > 0 {
                        Text("\(count)")
                            .font(.caption2)
                            .foregroundColor(.white)
                            .padding(4)
                            .background(Color.dangerRed)
                            .clipShape(Circle())
                            .offset(x: 18, y: -18)
                    }
                }
                
                // Информация
                VStack(alignment: .leading, spacing: Spacing.xs) {
                    Text(title)
                        .font(.bodyBold)
                        .foregroundColor(.textPrimary)
                    
                    // Метрики (максимум 2)
                    ForEach(Array(metrics.prefix(2)), id: \.0) { key, value in
                        HStack(spacing: Spacing.xs) {
                            Text(key)
                                .font(.caption)
                                .foregroundColor(.textSecondary)
                            Text(value)
                                .font(.captionBold)
                                .foregroundColor(.textPrimary)
                        }
                    }
                }
                
                Spacer()
                
                // Стрелка
                Image(systemName: "chevron.right")
                    .foregroundColor(.textSecondary)
            }
            .padding(Spacing.m)
            .background(cardBackground)
            .cornerRadius(CornerRadius.medium)
        }
        .buttonStyle(PlainButtonStyle())
    }
}
```

### **Селектор пользователя:**

```swift
struct UserSelectorView: View {
    @Binding var selectedUserId: String?
    let users: [UserOption]
    let currentUserId: String
    
    var body: some View {
        VStack(alignment: .leading, spacing: Spacing.s) {
            Text(localizationManager.localized("reports_select_user"))
                .font(.body)
                .foregroundColor(.textSecondary)
            
            ScrollView(.horizontal, showsIndicators: false) {
                HStack(spacing: Spacing.s) {
                    // "Я"
                    UserChip(
                        id: currentUserId,
                        name: localizationManager.localized("reports_current_user"),
                        icon: "person.fill",
                        isSelected: selectedUserId == currentUserId || selectedUserId == nil,
                        onTap: { selectedUserId = currentUserId }
                    )
                    
                    // Дети
                    ForEach(users) { user in
                        UserChip(
                            id: user.id,
                            name: user.name,
                            icon: user.avatar ?? "person.fill",
                            isSelected: selectedUserId == user.id,
                            onTap: { selectedUserId = user.id }
                        )
                    }
                }
                .padding(.horizontal, Spacing.screenPadding)
            }
        }
    }
}

struct UserChip: View {
    let id: String
    let name: String
    let icon: String
    let isSelected: Bool
    let onTap: () -> Void
    
    var body: some View {
        Button(action: onTap) {
            HStack(spacing: Spacing.xs) {
                Image(systemName: icon)
                    .font(.caption)
                
                Text(name)
                    .font(.body)
            }
            .foregroundColor(isSelected ? .white : .textPrimary)
            .padding(.horizontal, Spacing.m)
            .padding(.vertical, Spacing.s)
            .background(
                Capsule()
                    .fill(isSelected ? Color.primaryBlue : Color.backgroundMedium)
            )
        }
    }
}
```

## 📊 ВЗАИМОСВЯЗИ

### **Аналитика ↔ Отчеты:**
- Карточки компонентов в Аналитике → Модальные окна отчетов
- Краткая статистика в карточках обновляется из API

### **Уведомления → Отчеты:**
- Уведомление о новой утечке → DarkWebMonitoringModal (с выделением)
- Уведомление о нарушении → DrivingReportsModal (с выделением)
- Уведомление о попытке кражи → IdentityTheftModal (с выделением)

### **Настройки → Отчеты:**
- Расширенные настройки → Компонент → "Просмотреть отчеты" → Модал
- Родительский контроль → Ребенок → "Отчеты" → DrivingReportsModal / AICategoriesModal

## 🚀 ПРОИЗВОДИТЕЛЬНОСТЬ

### **Оптимизации:**

1. **Ленивая загрузка:**
   - Загружать данные только при открытии модального окна
   - Не загружать все отчеты сразу

2. **Кэширование:**
   - Использовать CacheManager для кэширования отчетов
   - TTL: 5 минут для отчетов, 1 час для статистики

3. **Пагинация:**
   - Загружать данные порциями (по 20 элементов)
   - Infinite scroll для больших списков

4. **Параллельная загрузка:**
   - Загружать статистику для всех компонентов параллельно
   - Использовать `withThrowingTaskGroup`
