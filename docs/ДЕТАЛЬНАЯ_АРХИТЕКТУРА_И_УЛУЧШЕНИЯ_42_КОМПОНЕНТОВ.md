## 2️⃣ ПРОИЗВОДИТЕЛЬНОСТЬ: ЛЕНИВАЯ ЗАГРУЗКА И КЭШИРОВАНИЕ

### ❌ ПРОБЛЕМА:
Загрузка статуса всех 42 компонентов замедляет экран.

### ✅ РЕШЕНИЕ: Ленивая загрузка + кэширование

**Пример реализации:**

```swift
// Core/Services/ComponentStatusService.swift
import Foundation
import Combine

class ComponentStatusService {
    static let shared = ComponentStatusService()
    
    private let cache = ComponentStatusCache.shared
    private let apiService = APIService.shared
    private var cancellables = Set<AnyCancellable>()
    
    // Кэш статусов (TTL: 5 минут)
    private var statusCache: [String: ComponentStatus] = [:]
    private var cacheTimestamp: [String: Date] = [:]
    private let cacheTTL: TimeInterval = 300 // 5 минут
    
    // MARK: - Ленивая загрузка статуса
    
    func getStatus(
        for componentId: String,
        priority: ComponentPriority = .normal
    ) -> AnyPublisher<ComponentStatus, Error> {
        // 1. Проверяем кэш
        if let cachedStatus = getCachedStatus(for: componentId) {
            return Just(cachedStatus)
                .setFailureType(to: Error.self)
                .eraseToAnyPublisher()
        }
        
        // 2. Загружаем из API (только если нет в кэше или кэш устарел)
        return apiService.getComponentStatus(componentId: componentId)
            .handleEvents(receiveOutput: { [weak self] status in
                // 3. Сохраняем в кэш
                self?.cacheStatus(status, for: componentId)
            })
            .eraseToAnyPublisher()
    }
    
    // MARK: - Batch загрузка (только критичных)
    
    func loadCriticalComponentsStatus() -> AnyPublisher<[ComponentStatus], Error> {
        let criticalComponents = [
            "crash_detection",
            "emergency_response",
            "self_harm_detection",
            "grooming_detection",
            "online_predators"
        ]
        
        return Publishers.MergeMany(
            criticalComponents.map { componentId in
                getStatus(for: componentId, priority: .critical)
            }
        )
        .collect()
        .eraseToAnyPublisher()
    }
    
    // MARK: - Кэширование
    
    private func getCachedStatus(for componentId: String) -> ComponentStatus? {
        guard let timestamp = cacheTimestamp[componentId],
              Date().timeIntervalSince(timestamp) < cacheTTL,
              let status = statusCache[componentId] else {
            return nil
        }
        return status
    }
    
    private func cacheStatus(_ status: ComponentStatus, for componentId: String) {
        statusCache[componentId] = status
        cacheTimestamp[componentId] = Date()
        
        // Сохраняем в UserDefaults для персистентности
        cache.saveStatus(status, for: componentId)
    }
    
    // MARK: - Background refresh
    
    func refreshStatusInBackground(for componentId: String) {
        DispatchQueue.global(qos: .background).async { [weak self] in
            _ = self?.getStatus(for: componentId)
                .sink(
                    receiveCompletion: { _ in },
                    receiveValue: { _ in }
                )
        }
    }
}

// Использование в ViewModel:
class VPNScreenViewModel: ObservableObject {
    @Published var crashDetectionEnabled = false
    @Published var isLoading = false
    
    private let statusService = ComponentStatusService.shared
    private var cancellables = Set<AnyCancellable>()
    
    func loadCriticalComponents() {
        isLoading = true
        
        statusService.loadCriticalComponentsStatus()
            .receive(on: DispatchQueue.main)
            .sink(
                receiveCompletion: { [weak self] completion in
                    self?.isLoading = false
                    if case .failure(let error) = completion {
                        print("Ошибка загрузки: \(error)")
                    }
                },
                receiveValue: { [weak self] statuses in
                    // Обновляем только критичные компоненты
                    for status in statuses {
                        self?.updateComponentStatus(status)
                    }
                    self?.isLoading = false
                }
            )
            .store(in: &cancellables)
    }
    
    func loadComponentOnDemand(_ componentId: String) {
        statusService.getStatus(for: componentId)
            .receive(on: DispatchQueue.main)
            .sink(
                receiveCompletion: { _ in },
                receiveValue: { [weak self] status in
                    self?.updateComponentStatus(status)
                }
            )
            .store(in: &cancellables)
    }
    
    private func updateComponentStatus(_ status: ComponentStatus) {
        switch status.componentId {
        case "crash_detection":
            crashDetectionEnabled = status.isEnabled
        // ... остальные компоненты
        default:
            break
        }
    }
}
```

**Преимущества:**
- ✅ Быстрая загрузка экрана (только критичные компоненты)
- ✅ Кэширование снижает нагрузку на API
- ✅ Background refresh обновляет данные незаметно
- ✅ Ленивая загрузка по требованию

```swift
// ViewModels/VPNScreenViewModel.swift
import Foundation
import Combine
import SwiftUI

@MainActor
class VPNScreenViewModel: ObservableObject {
    // MARK: - Published Properties
    
    @Published var crashDetectionEnabled = false
    @Published var roadsideAssistanceEnabled = false
    @Published var incidentResponseEnabled = false
    @Published var passwordSecurityEnabled = false
    @Published var phishingProtectionEnabled = false
    @Published var malwareDetectionEnabled = false
    @Published var mobileSecurityEnabled = false
    @Published var networkSecurityEnabled = false
    @Published var emergencyResponseEnabled = false
    @Published var emergencyEventEnabled = false
    
    @Published var isLoading = false
    @Published var errorMessage: String?
    @Published var showPasswordGenerator = false
    @Published var showIncidentResponseSettings = false
    
    // MARK: - Dependencies
    
    private let apiService = APIService.shared
    private let statusService = ComponentStatusService.shared
    private let cacheService = ComponentCacheService.shared
    private var cancellables = Set<AnyCancellable>()
    
    // MARK: - Initialization
    
    init() {
        loadCriticalComponents()
    }
    
    // MARK: - Load Status
    
    func loadCriticalComponents() {
        isLoading = true
        
        statusService.loadCriticalComponentsStatus()
            .receive(on: DispatchQueue.main)
            .sink(
                receiveCompletion: { [weak self] completion in
                    self?.isLoading = false
                    if case .failure(let error) = completion {
                        self?.handleError(error)
                    }
                },
                receiveValue: { [weak self] statuses in
                    self?.updateStatuses(statuses)
                    self?.isLoading = false
                }
            )
            .store(in: &cancellables)
    }
    
    // MARK: - Toggle Functions
    
    func toggleCrashDetection() {
        let newValue = !crashDetectionEnabled
        
        // Оптимистичное обновление UI
        crashDetectionEnabled = newValue
        
        // API запрос
        if newValue {
            apiService.enableCrashDetection { [weak self] result in
                self?.handleToggleResult(result, componentId: "crash_detection")
            }
        } else {
            apiService.disableCrashDetection { [weak self] result in
                self?.handleToggleResult(result, componentId: "crash_detection")
            }
        }
    }
    
    func toggleRoadsideAssistance() {
        // Аналогично
    }
    
    // MARK: - Error Handling
    
    private func handleError(_ error: Error) {
        errorMessage = LocalizationManager.shared.localized("error_loading_status")
        ToastManager.shared.showError(error.localizedDescription)
    }
    
    private func handleToggleResult(_ result: Result<APIResponse<Bool>, Error>, componentId: String) {
        switch result {
        case .success(let response):
            if !response.success {
                // Откатываем оптимистичное обновление
                revertToggle(componentId: componentId)
                errorMessage = response.message ?? LocalizationManager.shared.localized("error_toggle_failed")
            }
        case .failure(let error):
            revertToggle(componentId: componentId)
            handleError(error)
        }
    }
    
    private func revertToggle(componentId: String) {
        // Откатываем изменение
        switch componentId {
        case "crash_detection":
            crashDetectionEnabled.toggle()
        // ... остальные
        default:
            break
        }
    }
    
    private func updateStatuses(_ statuses: [ComponentStatus]) {
        for status in statuses {
            switch status.componentId {
            case "crash_detection":
                crashDetectionEnabled = status.isEnabled
            case "roadside_assistance":
                roadsideAssistanceEnabled = status.isEnabled
            // ... остальные
            default:
                break
            }
        }
    }
}

// Использование в View:
struct VPNScreen: View {

    var body: some View {
        ScrollView {
            VStack(spacing: Spacing.m) {
                // Используем viewModel вместо @State
                SecurityFeatureRow(
                    name: LocalizationManager.shared.localized("crash_detection_title"),
                    isEnabled: $viewModel.crashDetectionEnabled,
                    onToggle: { viewModel.toggleCrashDetection() }
                )
            }
        }
        .onAppear {
            viewModel.loadCriticalComponents()
        }
    }
}
```

## 4️⃣ ОБРАБОТКА ОШИБОК: RETRY + TOAST

### ❌ ПРОБЛЕМА:
Недостаточная обработка ошибок сети.

### ✅ РЕШЕНИЕ: Retry механизм + Toast уведомления

**Пример реализации:**

```swift
// Shared/Utils/RetryManager.swift
import Foundation
import Combine

class RetryManager {
    static let shared = RetryManager()
    
    func retry<T>(
        maxAttempts: Int = 3,
        delay: TimeInterval = 2.0,
        operation: @escaping () -> AnyPublisher<T, Error>
    ) -> AnyPublisher<T, Error> {
        return operation()
            .catch { error -> AnyPublisher<T, Error> in
                if maxAttempts > 1 {
                    return Just(())
                        .delay(for: .seconds(delay), scheduler: DispatchQueue.main)
                        .flatMap { _ in
                            self.retry(
                                maxAttempts: maxAttempts - 1,
                                delay: delay * 1.5, // Экспоненциальная задержка
                                operation: operation
                            )
                        }
                        .eraseToAnyPublisher()
                } else {
                    return Fail(error: error).eraseToAnyPublisher()
                }
            }
            .eraseToAnyPublisher()
    }
}

// Shared/Utils/ToastManager.swift
import SwiftUI

class ToastManager: ObservableObject {
    static let shared = ToastManager()
    
    @Published var currentToast: ToastMessage?
    
    func showSuccess(_ message: String) {
        currentToast = ToastMessage(
            message: message,
            type: .success,
            duration: 2.0
        )
    }
    
    func showError(_ message: String) {
        currentToast = ToastMessage(
            message: message,
            type: .error,
            duration: 3.0
        )
    }
    
    func showInfo(_ message: String) {
        currentToast = ToastMessage(
            message: message,
            type: .info,
            duration: 2.0
        )
    }
}

struct ToastMessage: Identifiable {
    let id = UUID()
    let message: String
    let type: ToastType
    let duration: TimeInterval
    
    enum ToastType {
        case success, error, info
    }
}

// SwiftUI компонент для Toast:
struct ToastView: View {
    @ObservedObject var toastManager = ToastManager.shared
    
    var body: some View {
        VStack {
            Spacer()
            
            if let toast = toastManager.currentToast {
                HStack {
                    Image(systemName: toast.type.icon)
                        .foregroundColor(toast.type.color)
                    
                    Text(toast.message)
                        .font(.body)
                        .foregroundColor(.white)
                }
                .padding()
                .background(
                    RoundedRectangle(cornerRadius: CornerRadius.medium)
                        .fill(toast.type.backgroundColor)
                )
                .padding()
                .transition(.move(edge: .bottom).combined(with: .opacity))
                .onAppear {
                    DispatchQueue.main.asyncAfter(deadline: .now() + toast.duration) {
                        withAnimation {
                            toastManager.currentToast = nil
                        }
                    }
                }
            }
        }
    }
}

extension ToastMessage.ToastType {
    var icon: String {
        switch self {
        case .success: return "checkmark.circle.fill"
        case .error: return "xmark.circle.fill"
        case .info: return "info.circle.fill"
        }
    }
    
    var color: Color {
        switch self {
        case .success: return .green
        case .error: return .red
        case .info: return .blue
        }
    }
    
    var backgroundColor: Color {
        switch self {
        case .success: return .green.opacity(0.9)
        case .error: return .red.opacity(0.9)
        case .info: return .blue.opacity(0.9)
        }
    }
}

// Использование в ViewModel:
func toggleCrashDetection() {
    let newValue = !crashDetectionEnabled
    crashDetectionEnabled = newValue
    
    RetryManager.shared.retry(maxAttempts: 3) { [weak self] in
        guard let self = self else {
            return Fail(error: NSError(domain: "ViewModel", code: -1))
                .eraseToAnyPublisher()
        }
        
        if newValue {
            return self.apiService.enableCrashDetection()
                .map { $0.data ?? false }
                .eraseToAnyPublisher()
        } else {
            return self.apiService.disableCrashDetection()
                .map { $0.data ?? false }
                .eraseToAnyPublisher()
        }
    }
    .receive(on: DispatchQueue.main)
    .sink(
        receiveCompletion: { [weak self] completion in
            if case .failure(let error) = completion {
                self?.crashDetectionEnabled.toggle() // Откат
                ToastManager.shared.showError(
                    LocalizationManager.shared.localized("error_toggle_failed")
                )
            }
        },
        receiveValue: { [weak self] success in
            if success {
                ToastManager.shared.showSuccess(
                    LocalizationManager.shared.localized("component_enabled")
                )
            }
        }
    )
    .store(in: &cancellables)
}
```

```swift
// Расширение LocalizationManager:
extension LocalizationManager {
    func localized(_ key: String, args: [String] = []) -> String {
        var string = localized(key)
        for (index, arg) in args.enumerated() {
            string = string.replacingOccurrences(of: "%\(index)", with: arg)
        }
        return string
    }
}
```
