func callAssistance(problem: RoadsideProblem, description: String, vehicleInfo: VehicleInfo) {
    // Получить текущее местоположение
    LocationManager.shared.getCurrentLocation { [weak self] location in
        guard let location = location else { return }
        
        let request = RoadsideAssistanceRequest(
            location: location,
            problem: problem,
            vehicleInfo: vehicleInfo,
            description: description.isEmpty ? nil : description
        )
        
        apiService.callRoadsideAssistance(request: request) { result in
            DispatchQueue.main.async {
                switch result {
                case .success(let response):
                    // Показать статус запроса
                    self?.showRequestStatus(response)
                case .failure(let error):
                    // Показать ошибку
                    print("Ошибка: \(error)")
                }
            }
        }
    }
