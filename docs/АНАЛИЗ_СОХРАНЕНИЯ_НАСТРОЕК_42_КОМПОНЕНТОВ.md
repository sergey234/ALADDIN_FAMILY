## 📝 РЕКОМЕНДАЦИИ

### 1. Использовать единый подход к сохранению:

```swift
// Для простых настроек (Bool, String, Int)
@AppStorage("component_\(componentId)_setting_key") private var setting: Bool = false

// Для сложных настроек
private func saveSettings() {
    Task {
        do {
            let config = ComponentConfiguration(
                componentId: componentId,
                settings: [
                    "setting1": setting1,
                    "setting2": setting2
                ]
            )
            try await configurationService.saveConfiguration(
                componentId: componentId,
                configuration: config
            )
            // Сохранить локально
            UserDefaults.standard.set(setting1, forKey: "component_\(componentId)_setting1")
            toastManager.showSuccess("Настройки сохранены")
        } catch {
            toastManager.showError("Ошибка сохранения")
        }
    }
}
```

### 2. Загружать настройки при открытии:

```swift
.onAppear {
    loadSettings()
}

private func loadSettings() {
    // Загрузить из UserDefaults
    if let saved = UserDefaults.standard.object(forKey: "component_\(componentId)_setting1") as? Bool {
        setting1 = saved
    }
    
    // Загрузить из API (опционально)
    Task {
        do {
            let config = try await configurationService.getConfiguration(for: componentId)
            // Применить настройки из конфигурации
        } catch {
            // Использовать локальные настройки
        }
    }
}
```

### 3. Сохранять настройки при каждом изменении (для тумблеров):
