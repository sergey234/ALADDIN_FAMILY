```swift
    onToggle: { viewModel.togglePhishingProtection() }
)
```

#### ⚠️ ПРОБЛЕМА: `updateLocalStatuses()` может не обновлять статусы

**Текущий код:**
```swift
private func updateLocalStatuses() async {
    for componentId in componentIds {
        do {
            let status = try await statusService.getStatus(for: componentId)
            updateStatusForComponent(componentId: componentId, status: status)
        } catch {
            print("⚠️ Ошибка загрузки статуса для \(componentId): \(error)")
            // ❌ ПРОБЛЕМА: При ошибке статус НЕ обновляется!
        }
    }
}
```

**Проблема:**
- Если `getStatus()` выбрасывает ошибку → статус не обновляется
- Компоненты остаются с дефолтным значением `false`
- Но даже если они `false`, они должны **ОТОБРАЖАТЬСЯ** на экране!

#### ✅ РЕШЕНИЕ: Использовать fallback значения

**Исправленный код:**
```swift
private func updateLocalStatuses() async {
    for componentId in componentIds {
        do {
            let status = try await statusService.getStatus(for: componentId)
            updateStatusForComponent(componentId: componentId, status: status)
        } catch {
            // ✅ FALLBACK: Использовать дефолтное значение из componentStatuses
            if let defaultStatus = await statusService.componentStatuses[componentId] {
                updateStatusForComponent(componentId: componentId, status: defaultStatus)
            } else {
                // Если даже дефолтного нет, создать новый со значением false
                let fallbackStatus = ComponentStatus(
                    componentId: componentId,
                    isEnabled: false,
                    lastUpdate: nil,
                    configuration: nil
                )
                updateStatusForComponent(componentId: componentId, status: fallbackStatus)
            }
        }
    }
}
```
