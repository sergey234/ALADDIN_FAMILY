### ✅ ШАГ 2.3: Исправить TariffsViewModel для iPad

```swift
private func purchaseTariff(_ tariff: Tariff) async {
    // ✅ ДОБАВИТЬ: Проверка что StoreManager готов
    guard !storeManager.isLoading else {
        await MainActor.run {
            errorMessage = "Пожалуйста, подождите..."
        }
        return
    }
    
    // ✅ ДОБАВИТЬ: Проверка что продукт найден
    guard let product = storeManager.products.first(where: { $0.id == tariff.productID }) else {
        await MainActor.run {
            errorMessage = "Продукт не найден. Попробуйте позже."
        }
        // Попробовать загрузить продукты
        await storeManager.loadProducts()
        return
    }
    
    // ✅ ДОБАВИТЬ: Обработка ошибок с детальным логированием
    do {
        print("🔍 [TariffsViewModel] Starting purchase for: \(product.id)")
        let transaction = try await storeManager.purchase(product)
        if transaction != nil {
            await MainActor.run {
                errorMessage = nil
                isPurchaseSuccessful = true
                // Обновить UI
            }
            print("✅ [TariffsViewModel] Purchase successful")
        } else {
            await MainActor.run {
                errorMessage = "Покупка отменена"
            }
        }
    } catch {
        await MainActor.run {
            errorMessage = "Ошибка покупки: \(error.localizedDescription)"
        }
        print("❌ [TariffsViewModel] Purchase error: \(error)")
        print("❌ [TariffsViewModel] Error details: \(error)")
    }
}
```
