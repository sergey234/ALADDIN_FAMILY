"Учетная запись приложения Apple" - это раздел в App Store Connect,
где настраивается информация о подписках (Subscriptions).

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📍 ГДЕ НАХОДИТСЯ:

App Store Connect → Ваше приложение → Features → In-App Purchases
→ Subscription Group → Subscription

В разделе "Subscription Information" есть поле "Account Information"
или информация об учетной записи подписки.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📋 ЧТО ТАМ НУЖНО УКАЗЫВАТЬ:

1. SUBSCRIPTION GROUP (Группа подписок):
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   • Reference Name (внутреннее название группы)
     Пример: "ALADDIN Subscriptions"
   
   • Display Name (название для пользователей)
     Пример: "ALADDIN Protection Plans"

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

2. ДЛЯ КАЖДОЙ ПОДПИСКИ В ГРУППЕ:

   A) BASIC INFORMATION (Основная информация):
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   • Reference Name (внутреннее название)
     Пример: "ALADDIN Individual"
   
   • Product ID (уникальный идентификатор)
     Пример: "family.aladdin.ios.subscription.individual.v2"
     ⚠️ Должен совпадать с кодом в StoreManager.swift!
   
   • Subscription Duration (период подписки)
     Пример: 1 месяц / 1 year
   
   • Subscription Price (цена)
     Выбирается из прайс-листа Apple

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   B) LOCALIZATION (Локализация):
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   Для каждого языка (English, Russian и т.д.):

   • Display Name (отображаемое название)
     Пример (EN): "ALADDIN Individual"
     Пример (RU): "ALADDIN Индивидуальный"
   
   • Description (описание - максимум 55 символов!)
     Пример (EN): "Up to 1 device. Full protection & AI assistant."
     Пример (RU): "1 устройство. Полная защита и AI-помощник."
     
     ⚠️ ВАЖНО: Это поле, которое мы обсуждали!
     Именно здесь указывается краткое описание подписки.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   C) REVIEW INFORMATION (Информация для проверки):
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   • Review Notes (примечания для ревьюеров)
     Можно указать, как протестировать подписку
   
   • Screenshot (скриншот экрана с подпиской)
     Показывает, как подписка выглядит в приложении
