    async def check_email_breach(self, email: str) -> List[Dict]:
        """
        Проверка email на утечки через Have I Been Pwned API
        
        Использует k-анонимность для безопасности
        
        Args:
            email: Email адрес для проверки
            
        Returns:
            Список найденных утечек
        """
        try:
            # Валидация email (используем общую утилиту)
            if self.threat_intelligence_agent:
                if not self._validate_email(email):
                    self.logger.warning(f"Невалидный email: {email}")
                    return []
            
            # Используем k-анонимность для безопасности
            email_hash = hashlib.sha1(email.encode()).hexdigest()
            hash_prefix = email_hash[:5].upper()
            hash_suffix = email_hash[5:].upper()
            
            # Запрос к Have I Been Pwned API
            url = f"{self.haveibeenpwned_api}/range/{hash_prefix}"
            
            # Используем общий метод HTTP запроса (если есть)
            if self.threat_intelligence_agent:
                response = await self.threat_intelligence_agent._make_http_request(url)
            else:
                response = requests.get(url, timeout=10)
            
            if response.status_code == 200:
                breaches = self._parse_breaches(response.text, hash_suffix, email)
                return breaches
            elif response.status_code == 404:
                # Email не найден в утечках
                return []
            else:
                self.logger.error(f"Ошибка API Have I Been Pwned: {response.status_code}")
                return []
                
        except Exception as e:
            self.logger.error(f"Ошибка проверки email {email}: {e}")
            return []
    
    def _parse_breaches(self, response_text: str, hash_suffix: str, email: str) -> List[Dict]:
        """
        Парсинг ответа от Have I Been Pwned API
        
        Args:
            response_text: Текст ответа от API
            hash_suffix: Суффикс хеша для проверки
            email: Email адрес
            
        Returns:
            Список найденных утечек
        """
        breaches = []
        lines = response_text.strip().split('\n')
        
        for line in lines:
            if not line:
                continue
                
            parts = line.split(':')
            if len(parts) >= 2:
                hash_part = parts[0].upper()
                count = int(parts[1])
                
                # Проверяем, соответствует ли хеш нашему email
                if hash_part == hash_suffix:
                    # Найдена утечка - получаем детали через API
                    breach_name = parts[2] if len(parts) > 2 else "Unknown"
                    breaches.append({
                        "email": email,
                        "breach_name": breach_name,
                        "count": count,
                        "detected_at": datetime.now().isoformat()
                    })
        
        return breaches

            user_id=user_id,
            email=email,
            interval_hours=interval_hours
        ))
        
        return jsonify({"status": "monitoring_started"}), 200
        
    except Exception as e:
        return jsonify({"error": str(e)}), 500

@app.route('/api/darkweb/breaches', methods=['GET'])
async def get_breaches():
    """
    Получить список утечек для пользователя
    
    Query parameters:
    - user_id: string (required)
    """
    try:
        user_id = request.args.get('user_id')
        
        if not user_id:
            return jsonify({"error": "user_id is required"}), 400
        
        # Получить утечки из мониторинга
        if user_id in dark_web_agent.monitored_users:
            result = dark_web_agent.monitored_users[user_id]
            return jsonify({
                "breaches": result.get("breaches", []),
                "breaches_found": result.get("breaches_found", 0),
                "checked_at": result.get("checked_at")
            }), 200
        else:
            return jsonify({"breaches": [], "breaches_found": 0}), 200
        
    except Exception as e:
        return jsonify({"error": str(e)}), 500
