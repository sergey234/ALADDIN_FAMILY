**Задача:** Добавить метод `saveSettings()` через `ComponentConfigurationService`

**Решение:**
```swift
private func saveSettings() {
    Task {
        do {
            let isComponentEnabled = await MainActor.run {
                ComponentStatusService.shared.getComponentEnabledStatus(componentId: "content_blocker_manager")
            }
            
            let config = ComponentConfiguration(
                isEnabled: isComponentEnabled,
                priority: .normal,
                additionalSettings: [
                    "selectedCategories": AnyCodable(selectedCategories.map { $0.rawValue })
                ]
            )
            
            try await ComponentConfigurationService.shared.saveConfiguration(
                componentId: "content_blocker_manager",
                configuration: config
            )
            
            await MainActor.run {
                HapticFeedback.notification(.success)
                isPresented = false
            }
        } catch {
            // Fallback: сохранить локально
            if let encoded = try? JSONEncoder().encode(Array(selectedCategories)) {
                UserDefaults.standard.set(encoded, forKey: "content_blocker_selected_categories")
            }
            await MainActor.run {
                HapticFeedback.notification(.success)
                isPresented = false
            }
        }
    }
}
```

## 🔧 ПРАВИЛА ПРЕДОТВРАЩЕНИЯ ОШИБОК

### 1. Четкие правила использования

#### Правило 1: Простые настройки → `@AppStorage`
```swift
// ✅ ПРАВИЛЬНО:
@AppStorage("setting_key") private var isEnabled: Bool = false

// ❌ НЕПРАВИЛЬНО:
@State private var isEnabled: Bool = false  // Не сохраняется!
```

#### Правило 2: Компоненты 42 → `ComponentStatusService`
```swift
// ✅ ПРАВИЛЬНО:
try await statusService.updateStatus(
    componentId: componentId,
    isEnabled: newValue
)

// ❌ НЕПРАВИЛЬНО:
@State private var isEnabled: Bool = false  // Только UI состояние
```

#### Правило 3: Сложные настройки → `ComponentConfigurationService`
```swift
// ✅ ПРАВИЛЬНО:
let config = ComponentConfiguration(
    isEnabled: isEnabled,
    priority: .normal,
    additionalSettings: [
        "key1": AnyCodable(value1),
        "key2": AnyCodable(value2)
    ]
)
try await configurationService.saveConfiguration(
    componentId: componentId,
    configuration: config
)

// ❌ НЕПРАВИЛЬНО:
// ComponentConfiguration с componentId в инициализаторе
```
