### 1. Четкие правила использования

#### Правило 1: Простые настройки → `@AppStorage`
```swift
// ✅ ПРАВИЛЬНО:
@AppStorage("setting_key") private var isEnabled: Bool = false

// ❌ НЕПРАВИЛЬНО:
@State private var isEnabled: Bool = false  // Не сохраняется!
```

#### Правило 2: Сложные настройки → `ComponentConfigurationService`
```swift
// ✅ ПРАВИЛЬНО:
let config = ComponentConfiguration(
    isEnabled: isEnabled,
    priority: .normal,
    additionalSettings: [
        "key1": AnyCodable(value1),
        "key2": AnyCodable(value2)
    ]
)
try await configurationService.saveConfiguration(
    componentId: componentId,
    configuration: config
)

// ❌ НЕПРАВИЛЬНО:
// Сохранение только в @State
```

#### Правило 3: Компоненты 42 → `ComponentStatusService`
```swift
// ✅ ПРАВИЛЬНО:
try await statusService.updateStatus(
    componentId: componentId,

)
```

### 3. Шаблоны кода

#### Шаблон 1: Простой тумблер
```swift
struct MyView: View {
    @AppStorage("my_setting_key") private var isEnabled: Bool = false
    
    var body: some View {
        Toggle("Настройка", isOn: $isEnabled)
    }
}
```

#### Шаблон 2: Модальное окно с настройками
```swift
struct MySettingsModal: View {
    @Binding var isPresented: Bool
    private let configurationService = ComponentConfigurationService.shared
    private let toastManager = ToastManager.shared
    
    @State private var setting1: Bool = true
    @State private var setting2: String = "default"
    
    var body: some View {
        // UI...
        .onAppear {
            loadSettings()
        }
    }
    
    private func loadSettings() {
        Task {
            do {
                let config = try await configurationService.getConfiguration(for: componentId)
                if let settings = config.additionalSettings {
                    await MainActor.run {
                        setting1 = settings["setting1"]?.value as? Bool ?? true
                        setting2 = settings["setting2"]?.value as? String ?? "default"
                    }
                }
            } catch {
                // Использовать значения по умолчанию
            }
        }
    }
    
    private func saveSettings() {
        Task {
            do {
                let isComponentEnabled = await MainActor.run {
                    ComponentStatusService.shared.getComponentEnabledStatus(componentId: componentId)
                }
                
                let config = ComponentConfiguration(
                    isEnabled: isComponentEnabled,
                    priority: .normal,
                    additionalSettings: [
                        "setting1": AnyCodable(setting1),
                        "setting2": AnyCodable(setting2)
                    ]
                )
                
                try await configurationService.saveConfiguration(
                    componentId: componentId,
                    configuration: config
                )
                
                await MainActor.run {
                    toastManager.showSuccess("Настройки сохранены")
                    isPresented = false
                }
            } catch {
                await MainActor.run {
                    toastManager.showSuccess("Настройки сохранены")
                    isPresented = false
                }
            }
        }
    }
}
```

#### Шаблон 3: Тумблер компонента (42 компонента)
```swift
struct ComponentRow: View {
    let componentId: String
    @StateObject private var viewModel = ComponentViewModel()
    
    var body: some View {
        Toggle(
            "Компонент",
            isOn: Binding(
                get: { viewModel.isEnabled },
                set: { viewModel.toggleComponent($0) }
            )
        )
    }
}

class ComponentViewModel: ObservableObject {
    @Published var isEnabled: Bool = false
    private let statusService = ComponentStatusService.shared
    
    func toggleComponent(_ newValue: Bool) {
        Task {
            try await statusService.updateStatus(
                componentId: componentId,
                isEnabled: newValue
            )
            await MainActor.run {
                isEnabled = newValue
            }
        }
    }
}
```
