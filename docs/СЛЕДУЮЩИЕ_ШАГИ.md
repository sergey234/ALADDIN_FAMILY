  -H "Authorization: token $GITHUB_TOKEN" \

  -d '{"ref":"master"}'

  python3 -c "import sys, json; r=json.load(sys.stdin)['workflow_runs'][0]; print(f'#{r[\"run_number\"]}'); print(r['html_url'])"
