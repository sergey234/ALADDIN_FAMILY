## 💡 РЕКОМЕНДАЦИИ ПО РЕАЛИЗАЦИИ

### 1. Единый подход к сохранению настроек:

```swift
// В модальном окне
@StateObject private var configurationService = ComponentConfigurationService.shared
@StateObject private var toastManager = ToastManager.shared

// Для каждого параметра добавить @AppStorage
@AppStorage("component_\(componentId)_setting_key") private var setting: Bool = false

// В saveSettings()
private func saveSettings() {
    Task {
        do {
            // 1. Сохранить локально (уже через @AppStorage)
            
            // 2. Создать конфигурацию
            let config = ComponentConfiguration(
                componentId: componentId,
                settings: [
                    "setting1": setting1,
                    "setting2": setting2,
                    // ...
                ]
            )
            
            // 3. Сохранить через сервис
            try await configurationService.saveConfiguration(
                componentId: componentId,
                configuration: config
            )
            
            // 4. Показать успех
            await MainActor.run {
                toastManager.showSuccess("Настройки сохранены")
                isPresented = false
            }
        } catch {
            await MainActor.run {
                toastManager.showError("Ошибка сохранения")
            }
        }
    }
}

// Загрузка при открытии
.onAppear {
    loadSettings()
}

private func loadSettings() {
    // Настройки уже загружены через @AppStorage
    // Но можно загрузить из API для синхронизации
    Task {
        do {
            let config = try await configurationService.getConfiguration(for: componentId)
            // Применить настройки из конфигурации
        } catch {
            // Использовать локальные настройки (@AppStorage)
        }
    }
}
```
