```python
# backend_agents/dark_web_monitoring_agent.py

from security.ai_agents.threat_intelligence_agent import ThreatIntelligenceAgent
from security.base import SecurityBase
from typing import Optional, Dict, Any
import hashlib

class DarkWebMonitoringAgent(SecurityBase):
    """Агент для мониторинга утечек в Dark Web (гибридный подход)"""
    
    def __init__(self, config: Optional[Dict[str, Any]] = None):
        super().__init__(config)
        # Используем утилиты из ThreatIntelligenceAgent
        self.threat_intel = ThreatIntelligenceAgent()
        self.hibp_api_key = config.get("hibp_api_key", "")
        
    def check_email_breach(self, email: str) -> dict:
        """Проверка email на утечки с k-анонимностью"""
        # Используем общий метод валидации
        if not self._validate_email(email):
            return {"error": "Invalid email"}
        
        # k-анонимность для безопасности
        email_hash = self._hash_email_k_anonymous(email)
        
        # Используем общий HTTP клиент
        response = self._make_http_request(
            f"https://api.haveibeenpwned.com/v3/breachedaccount/{email_hash[:5]}",
            headers={"hibp-api-key": self.hibp_api_key}
        )
        
        return self._parse_breach_response(response, email)
    
    def _hash_email_k_anonymous(self, email: str) -> str:
        """k-анонимность для безопасности email"""
        return hashlib.sha1(email.encode()).hexdigest().upper()
    
    def _validate_email(self, email: str) -> bool:
        """Используем общий метод валидации"""
        return self.threat_intel._validate_email(email)
    
    def _make_http_request(self, url: str, headers: dict = None) -> dict:
        """Используем общий HTTP клиент"""
        return self.threat_intel._make_http_request(url, headers)
```

from security.compliance.russian_data_protection_manager import RussianDataProtectionManager
from security.base import SecurityBase

class RussianIdentityTheftProtectionAgent(SecurityBase):
    """Агент для защиты от кражи личности в России"""
    
    def __init__(self, config: Optional[Dict[str, Any]] = None):
        super().__init__(config)
        self.data_protection = RussianDataProtectionManager()
        
    def monitor_snils(self, snils: str) -> dict:
        """Мониторинг СНИЛС на подозрительную активность"""
        # Шифрование СНИЛС (AES-256)
        encrypted_snils = self.data_protection.encrypt_data(snils)
        
        # Проверка в базе мошенников
        fraud_check = self._check_fraud_database(encrypted_snils)
        
        return {
            "snils": encrypted_snils,
            "fraud_detected": fraud_check["found"],
            "risk_score": fraud_check["risk_score"]
        }
