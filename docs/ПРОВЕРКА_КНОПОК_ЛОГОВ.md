## 🔧 КАК ЭТО РАБОТАЕТ

### **Код:**

```javascript
let currentService = 'payment_service';  // Текущий тип логов
let autoRefreshInterval = null;          // Интервал автообновления

// Функция загрузки логов
async function loadLogs() {
  // Использует currentService для загрузки нужного типа логов
  const data = await window.adminAPI.apiRequest(
    `/api/admin/logs?service=${currentService}&limit=100`
  );
  // ... отображение логов
}

// Переключение типа логов
function selectService(service) {
  currentService = service;  // Обновляем текущий тип
  loadLogs();                // Загружаем логи
  // Автообновление продолжит работать с новым типом
}

// Автообновление
function toggleAutoRefresh() {
  if (autoRefreshInterval) {
    clearInterval(autoRefreshInterval);  // Останавливаем
  } else {
    autoRefreshInterval = setInterval(loadLogs, 5000);  // Запускаем
    // loadLogs() использует currentService
  }
}
```
