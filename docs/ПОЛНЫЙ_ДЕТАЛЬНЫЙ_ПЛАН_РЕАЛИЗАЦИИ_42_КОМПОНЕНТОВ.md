```swift
// ViewModels/EmergencyAssistanceViewModel.swift
import Foundation
import Combine

@MainActor
class EmergencyAssistanceViewModel: ObservableObject {
    @Published var crashDetectionEnabled = false
    @Published var roadsideAssistanceEnabled = false
    @Published var emergencyContacts: [EmergencyContact] = []
    @Published var emergencyHistory: [EmergencyEvent] = []
    @Published var crashDetectionStatus: String = "Неактивно"
    @Published var roadsideAssistanceStatus: String = "Неактивно"
    @Published var crashDetectionStatistics = CrashDetectionStatistics()
    @Published var showCallAssistance = false
    @Published var showAddContact = false
    @Published var selectedEvent: EmergencyEvent?
    
    private let apiService = APIService.shared
    private var cancellables = Set<AnyCancellable>()
    
    init() {
        loadStatus()
        loadContacts()
        loadHistory()
    }
    
    func toggleCrashDetection() {
        if crashDetectionEnabled {
            apiService.disableCrashDetection { [weak self] result in
                DispatchQueue.main.async {
                    switch result {
                    case .success(let response):
                        if response.success {
                            self?.crashDetectionEnabled = false
                            self?.crashDetectionStatus = "Неактивно"
                        }
                    case .failure(let error):
                        print("Ошибка отключения: \(error.localizedDescription)")
                    }
                }
            }
        } else {
            apiService.enableCrashDetection { [weak self] result in
                DispatchQueue.main.async {
                    switch result {
                    case .success(let response):
                        if response.success {
                            self?.crashDetectionEnabled = true
                            self?.crashDetectionStatus = "Активно"
                        }
                    case .failure(let error):
                        print("Ошибка включения: \(error.localizedDescription)")
                    }
                }
            }
        }
    }
    
    func toggleRoadsideAssistance() {
        if roadsideAssistanceEnabled {
            apiService.disableRoadsideAssistance { [weak self] result in
                DispatchQueue.main.async {
                    switch result {
                    case .success(let response):
                        if response.success {
                            self?.roadsideAssistanceEnabled = false
                            self?.roadsideAssistanceStatus = "Неактивно"
                        }
                    case .failure(let error):
                        print("Ошибка: \(error.localizedDescription)")
                    }
                }
            }
        } else {
            apiService.enableRoadsideAssistance { [weak self] result in
                DispatchQueue.main.async {
                    switch result {
                    case .success(let response):
                        if response.success {
                            self?.roadsideAssistanceEnabled = true
                            self?.roadsideAssistanceStatus = "Активно"
                        }
                    case .failure(let error):
                        print("Ошибка: \(error.localizedDescription)")
                    }
                }
            }
        }
    }
    
    func loadStatus() {
        apiService.getCrashDetectionStatus { [weak self] result in
            DispatchQueue.main.async {
                switch result {
                case .success(let response):
                    self?.crashDetectionEnabled = response.isEnabled
                    self?.crashDetectionStatus = response.isMonitoring ? "Мониторинг активен" : "Неактивно"
                    self?.crashDetectionStatistics = CrashDetectionStatistics(
                        totalCrashes: response.totalCrashesDetected,
                        lastCrash: response.lastCrashDetected
                    )
                case .failure(let error):
                    print("Ошибка загрузки статуса: \(error.localizedDescription)")
                }
            }
        }
        
        apiService.getRoadsideAssistanceStatus { [weak self] result in
            DispatchQueue.main.async {
                switch result {
                case .success(let response):
                    self?.roadsideAssistanceEnabled = response.isEnabled
                    self?.roadsideAssistanceStatus = response.isEnabled ? "Активно" : "Неактивно"
                case .failure(let error):
                    print("Ошибка: \(error.localizedDescription)")
                }
            }
        }
    }
    
    func loadContacts() {
        apiService.getEmergencyContacts { [weak self] result in
            DispatchQueue.main.async {
                switch result {
                case .success(let contacts):
                    self?.emergencyContacts = contacts
                case .failure(let error):
                    print("Ошибка загрузки контактов: \(error.localizedDescription)")
                }
            }
        }
    }
    
    func loadHistory() {
        apiService.getEmergencyHistory { [weak self] result in
            DispatchQueue.main.async {
                switch result {
                case .success(let history):
                    self?.emergencyHistory = history
                case .failure(let error):
                    print("Ошибка загрузки истории: \(error.localizedDescription)")
                }
            }
        }
    }
    
    func addContact(_ contact: EmergencyContact) {
        apiService.addEmergencyContact(contact: contact) { [weak self] result in
            DispatchQueue.main.async {
                switch result {
                case .success(let newContact):
                    self?.emergencyContacts.append(newContact)
                    self?.showAddContact = false
                case .failure(let error):
                    print("Ошибка добавления контакта: \(error.localizedDescription)")
                }
            }
        }
    }
    
    func editContact(_ contact: EmergencyContact) {
        apiService.updateEmergencyContact(contact: contact) { [weak self] result in
            DispatchQueue.main.async {
                switch result {
                case .success(let updatedContact):
                    if let index = self?.emergencyContacts.firstIndex(where: { $0.id == contact.id }) {
                        self?.emergencyContacts[index] = updatedContact
                    }
                case .failure(let error):
                    print("Ошибка обновления контакта: \(error.localizedDescription)")
                }
            }
        }
    }
    
    func deleteContact(_ contact: EmergencyContact) {
        apiService.deleteEmergencyContact(contactId: contact.id) { [weak self] result in
            DispatchQueue.main.async {
                switch result {
                case .success:
                    self?.emergencyContacts.removeAll { $0.id == contact.id }
                case .failure(let error):
                    print("Ошибка удаления контакта: \(error.localizedDescription)")
                }
            }
        }
    }
    
    func callAssistance(problem: RoadsideProblem, description: String, vehicleInfo: VehicleInfo) {
        LocationManager.shared.getCurrentLocation { [weak self] location in
            guard let location = location else {
                print("Ошибка получения местоположения")
                return
            }
            
            let request = RoadsideAssistanceRequest(
                location: location,
                problem: problem,
                vehicleInfo: vehicleInfo,
                description: description.isEmpty ? nil : description
            )
            
            self?.apiService.callRoadsideAssistance(request: request) { result in
                DispatchQueue.main.async {
                    switch result {
                    case .success(let response):
                        // Добавить в историю
                        let event = EmergencyEvent(
                            id: response.requestId,
                            type: .roadside,
                            status: .active,
                            createdAt: Date(),
                            location: location,
                            description: description,
                            contactsNotified: []
                        )
                        self?.emergencyHistory.insert(event, at: 0)
                        self?.showCallAssistance = false
                    case .failure(let error):
                        print("Ошибка вызова помощи: \(error.localizedDescription)")
                    }
                }
            }
        }
    }
}

struct CrashDetectionStatistics {
    var totalCrashes: Int = 0
    var lastCrash: Date? = nil
}
```

```swift
import Foundation

// MARK: - Crash Detection Models

struct CrashDetectionStatusResponse: Codable {
    let isEnabled: Bool
    let isMonitoring: Bool
    let lastCrashDetected: Date?
    let totalCrashesDetected: Int
    let sensitivity: Double
}

struct CrashDetectionSettings: Codable {
    let sensitivity: Double // 0.0 - 1.0
    let autoCallEmergency: Bool
    let notifyContacts: Bool
    let notifyEmergencyServices: Bool
}

struct CrashEvent: Codable, Identifiable {
    let id: String
    let detectedAt: Date
    let location: Location
    let gForce: Double
    let severity: CrashSeverity
    let contactsNotified: [String]
    let emergencyCalled: Bool
}

enum CrashSeverity: String, Codable {
    case low = "low"
    case medium = "medium"
    case high = "high"
    case critical = "critical"
}

// MARK: - Roadside Assistance Models

struct RoadsideAssistanceRequest: Codable {
    let location: Location
    let problem: RoadsideProblem
    let vehicleInfo: VehicleInfo?
    let description: String?
}

enum RoadsideProblem: String, Codable {
    case flatTire = "flat_tire"
    case battery = "battery"
    case fuel = "fuel"
    case engine = "engine"
    case accident = "accident"
    case other = "other"
    
    var displayName: String {
        switch self {
        case .flatTire: return "Прокол колеса"
        case .battery: return "Разрядился аккумулятор"
        case .fuel: return "Закончилось топливо"
        case .engine: return "Проблема с двигателем"
        case .accident: return "Авария"
        case .other: return "Другое"
        }
    }
}

struct VehicleInfo: Codable {
    let make: String?
    let model: String?
    let year: Int?
    let licensePlate: String?
    let color: String?
}

struct RoadsideAssistanceResponse: Codable {
    let requestId: String
    let status: RequestStatus
    let estimatedArrival: Date?
    let serviceProvider: String?
    let contactPhone: String?
}
```

```swift
import SwiftUI

// MARK: - Emergency Feature Card

struct EmergencyFeatureCard: View {
    let icon: String
    let title: String
    let description: String
    @Binding var isEnabled: Bool
    let onToggle: () -> Void
    let status: String?
    let statistics: CrashDetectionStatistics?
    let actionButton: Button<Text>?
    
    init(
        icon: String,
        title: String,
        description: String,
        isEnabled: Binding<Bool>,
        onToggle: @escaping () -> Void,
        status: String? = nil,
        statistics: CrashDetectionStatistics? = nil,
        actionButton: Button<Text>? = nil
    ) {
        self.icon = icon
        self.title = title
        self.description = description
        self._isEnabled = isEnabled
        self.onToggle = onToggle
        self.status = status
        self.statistics = statistics
        self.actionButton = actionButton
    }
    
    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            HStack {
                Text(icon)
                    .font(.system(size: 32))
                
                VStack(alignment: .leading, spacing: 4) {

                Spacer()
                
                ALADDINToggle(isOn: $isEnabled, size: 40)
                    .onChange(of: isEnabled) { _ in
                        onToggle()
                    }
            }
            
            if isEnabled {
                if let status = status {
                    HStack {
                        Image(systemName: "checkmark.circle.fill")
                            .foregroundColor(.green)

                            .font(.system(size: 12))
                            .foregroundColor(.white.opacity(0.8))
                    }
                }
                
                if let statistics = statistics {
                    HStack(spacing: 16) {

        )
```
