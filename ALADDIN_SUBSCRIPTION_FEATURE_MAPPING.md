func shouldShowFeature(_ feature: AppFeature) -> Bool {

func performAIFunction() async throws {

#### **3. API LEVEL GATING:**
```python
@app.middleware("http")
async def check_subscription_access(request: Request, call_next):
    # Extract token from header
    token = request.headers.get("authorization")
    if not token:
        return JSONResponse(status_code=401, content={"error": "No token"})

    # Decode and validate subscription
    payload = decode_jwt_token(token)
    subscription = payload.get("subscription", {})

    # Check feature access
    endpoint = request.url.path
    if not has_feature_access(subscription, endpoint):
        return JSONResponse(status_code=403, content={"error": "Subscription required"})

    # Check usage limits
    if not check_rate_limits(subscription, endpoint):
        return JSONResponse(status_code=429, content={"error": "Rate limit exceeded"})

    return await call_next(request)
```
