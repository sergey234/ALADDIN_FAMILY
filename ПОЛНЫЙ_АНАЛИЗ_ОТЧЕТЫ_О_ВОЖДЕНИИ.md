```swift
func loadUsers() async {
    // 1. Проверяем токен
    guard AppConfig.authToken != nil else {
        viewModel.errorMessage = "Требуется авторизация..."
        return
    }
    
    // 2. Загружаем членов семьи из API
    do {
        let familyMembers = try await APIService.shared.getFamilyMembers()
        
        // 3. Преобразуем в список пользователей
        users = familyMembers.map { member in
            UserSelectorView.UserOption(
                id: member.id.uuidString,
                name: member.name ?? "Unknown"
            )
        }
        
        // 4. Устанавливаем currentUserId
        if let currentUser = familyMembers.first(where: { $0.isCurrentUser }) {
            currentUserId = currentUser.id.uuidString
        } else if let firstMember = familyMembers.first {
            currentUserId = firstMember.id.uuidString
        } else {
            currentUserId = "current"
        }
        
    } catch {
        // 5. Обработка ошибок
        let networkError = NetworkError.from(error)
        
        if case .unauthorized = networkError {
            // Graceful degradation - загружаем из UserDefaults
        }
        
        // Fallback на UserDefaults
        currentUserId = "current"
    }
    
    // 6. Синхронизация selectedUserId
    if selectedUserId.isEmpty && !currentUserId.isEmpty {
        selectedUserId = currentUserId
    }
}
```

**Файл:** `ViewModels/DrivingReportsViewModel.swift`

```swift
func loadReports(userId: String?, period: String) async {
    isLoading = true
    errorMessage = nil
    
    do {
        // 1. Загружаем статистику и отчеты параллельно
        async let statsTask = apiService.getDrivingStats(userId: userId, period: period)
        async let reportsTask = apiService.getDrivingReports(userId: userId, period: period)
        
        let (stats, reports) = try await (statsTask, reportsTask)
        
        // 2. Обновляем данные
        self.stats = stats
        self.reports = reports
        errorMessage = nil
        
    } catch {
        // 3. Обработка ошибок
        let networkError = NetworkError.from(error)
        
        // 3.1 Ошибка авторизации
        if case .unauthorized = networkError {
            errorMessage = "Требуется авторизация. Войдите в аккаунт для просмотра отчетов."
            return
        }
        
        // 3.2 Нет данных (404) - это нормально
        if case .notFound = networkError {
            self.stats = nil
            self.reports = []
            errorMessage = nil
            return
        }
        
        // 3.3 Другие ошибки
        if networkError.isCritical {
            errorMessage = String(format: "Не удалось загрузить отчет: %@", networkError.localizedDescription)
        } else {
            errorMessage = nil
        }
        
        self.stats = nil
        self.reports = []
    }
    
    isLoading = false
}
```

**Файл:** `app/security/api/routers/driving_reports_router.py`

**Логика:**
```python
@router.get("", response_model=List[DrivingReport])
def get_driving_reports(
    userId: Optional[str] = Query(None),
    period: str = Query("week", regex="^(week|month|year)$"),
    db: Session = Depends(get_db)
):
    try:
        # 1. Вычисляем дату начала периода
        now = datetime.utcnow()
        if period == "week":
            start_date = now - timedelta(days=7)
        elif period == "month":
            start_date = now - timedelta(days=30)
        else:  # year
            start_date = now - timedelta(days=365)
        
        # 2. Формируем SQL запрос
        query = text("""
            SELECT id, user_id, user_name, start_time, end_time,
                   start_location, end_location, distance, duration,
                   average_speed, max_speed, safety_score,
                   events, violations, positioning_system
            FROM driving_reports
            WHERE start_time >= :start_date
        """)
        
        params = {"start_date": start_date}
        
        # 3. Фильтр по userId если указан
        if userId:
            query = text("""
                SELECT ...
                FROM driving_reports
                WHERE start_time >= :start_date AND user_id = :user_id
                ORDER BY start_time DESC
            """)
            params["user_id"] = userId
        
        # 4. Выполняем запрос
        result = db.execute(query, params)
        rows = result.fetchall()
        
        # 5. Преобразуем результаты в модели
        reports = []
        for row in rows:
            # Парсим JSON поля если они есть
            events = row[12] if row[12] else []
            violations = row[13] if row[13] else []
            
            # Если это строки JSON, парсим их
            if isinstance(events, str):
                import json
                try:
                    events = json.loads(events)
                except:
                    events = []
            
            report = DrivingReport(
                id=str(row[0]),
                userId=str(row[1]),
                userName=str(row[2]) if row[2] else "Unknown",
                startTime=row[3],
                endTime=row[4],
                startLocation=str(row[5]) if row[5] else "",
                endLocation=str(row[6]) if row[6] else "",
                distance=float(row[7]) if row[7] else 0.0,
                duration=float(row[8]) if row[8] else 0.0,
                averageSpeed=float(row[9]) if row[9] else 0.0,
                maxSpeed=float(row[10]) if row[10] else 0.0,
                safetyScore=float(row[11]) if row[11] else 0.0,
                events=events if isinstance(events, list) else [],
                violations=violations if isinstance(violations, list) else [],
                positioningSystem=str(row[14]) if row[14] else None
            )
            reports.append(report)
        
        return reports
        
    except Exception as e:
        # 6. Graceful degradation
        print(f"⚠️ Ошибка получения отчетов из БД: {e}")
        return []  # Возвращаем пустой список
```
